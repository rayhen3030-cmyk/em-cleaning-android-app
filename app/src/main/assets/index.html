function serviceChanged() { const service = document.getElementById('service').value; const fensterBlock = document.getElementById('fensterBlock'); const flaechenBlock = document.getElementById('flaechenBlock'); fensterBlock.style.display = service === 'fenster' ? 'block' : 'none'; flaechenBlock.style.display = service === 'fenster' ? 'none' : 'block'; // Unterhaltsreinigung if (service === 'unterhalt') { document.getElementById('stundensatz').value = '30'; // Stunden werden automatisch berechnet document.getElementById('stunden').value = ''; document.getElementById('stunden').placeholder = 'Wird automatisch berechnet'; zeigeUnterhaltOptionen(); } else { document.getElementById('stunden').placeholder = 'Geschätzte Arbeitsstunden'; entferneUnterhaltOptionen(); } if (service === 'bau' || service === 'grund') { document.getElementById('stundensatz').value = '35'; } } function zeigeUnterhaltOptionen() { if (document.getElementById('unterhaltOptionen')) { return; } const block = document.createElement('div'); block.id = 'unterhaltOptionen'; block.innerHTML = `
Die App berechnet automatisch, wie viele Arbeitsstunden für die Fläche benötigt werden.

Beispiel:
500 m² ÷ 180 m²/Stunde = ca. 2,78 Stunden pro Reinigung.
`; const flaechenBlock = document.getElementById('flaechenBlock'); flaechenBlock.insertAdjacentElement( 'afterend', block ); } function entferneUnterhaltOptionen() { const block = document.getElementById('unterhaltOptionen'); if (block) { block.remove(); } } function preisBerechnen() { const service = document.getElementById('service').value; const stundensatz = parseFloat( document.getElementById('stundensatz').value ) || 0; const qmPreis = parseFloat( document.getElementById('qmPreis').value ) || 0; const anfahrt = parseFloat( document.getElementById('anfahrt').value ) || 0; const material = parseFloat( document.getElementById('material').value ) || 0; const zuschlagProzent = parseFloat( document.getElementById('zuschlag').value ) || 0; const mwst = parseFloat( document.getElementById('mwst').value ) || 0; let flaeche = 0; // ========================================== // FENSTER // ========================================== if (service === 'fenster') { const anzahl = parseFloat( document.getElementById('fensterAnzahl').value ) || 0; const breite = parseFloat( document.getElementById('fensterBreite').value ) || 0; const hoehe = parseFloat( document.getElementById('fensterHoehe').value ) || 0; const seiten = parseFloat( document.getElementById('fensterSeiten').value ) || 1; flaeche = anzahl * breite * hoehe * seiten; } else { flaeche = parseFloat( document.getElementById('qm').value ) || 0; } // ========================================== // UNTERHALTSREINIGUNG // ========================================== if (service === 'unterhalt') { if (flaeche <= 0) { alert( 'Bitte die Reinigungsfläche in m² eingeben.' ); return; } const leistung = parseFloat( document.getElementById( 'leistungProStunde' ).value ) || 180; const tageProWoche = parseFloat( document.getElementById( 'reinigungProWoche' ).value ) || 1; if (leistung <= 0) { alert( 'Bitte eine gültige Reinigungsleistung eingeben.' ); return; } // Benötigte Stunden für EINEN Einsatz const stundenProEinsatz = flaeche / leistung; // Durchschnitt: // 52 Wochen / 12 Monate const wochenProMonat = 52 / 12; const einsaetzeMonat = tageProWoche * wochenProMonat; const stundenMonat = stundenProEinsatz * einsaetzeMonat; // Lohn / Arbeitskosten const arbeitskosten = stundenMonat * stundensatz; // Anfahrt wird PRO EINSATZ gerechnet const anfahrtMonat = anfahrt * einsaetzeMonat; // Material als monatlicher Betrag const materialMonat = material; let netto = arbeitskosten + anfahrtMonat + materialMonat; const zuschlagEuro = netto * zuschlagProzent / 100; netto += zuschlagEuro; const steuer = netto * mwst / 100; const brutto = netto + steuer; // Stundenfeld automatisch füllen document.getElementById('stunden').value = stundenProEinsatz.toFixed(2); const box = document.getElementById('preisErgebnis'); box.style.display = 'block'; box.innerHTML = ` Unterhaltsreinigung

Reinigung: ${tageProWoche}× pro Woche
Ø Einsätze pro Monat: ${einsaetzeMonat.toFixed(1)}

Reinigungsfläche: ${flaeche.toFixed(2)} m²
Reinigungsleistung: ${leistung.toFixed(0)} m² / Std.

Stunden pro Reinigung: ${stundenProEinsatz.toFixed(2)} Std.
Arbeitsstunden pro Monat: ${stundenMonat.toFixed(2)} Std.

Stundensatz: ${stundensatz.toFixed(2)} €
Arbeitskosten monatlich: ${arbeitskosten.toFixed(2)} €
Anfahrt monatlich: ${anfahrtMonat.toFixed(2)} €
Material: ${materialMonat.toFixed(2)} €
Zuschlag: ${zuschlagEuro.toFixed(2)} €

Monatlich Netto: ${netto.toFixed(2)} €
MwSt.: ${steuer.toFixed(2)} €
${brutto .toFixed(2) .replace('.', ',')} € / Monat
`; return; } // ========================================== // ANDERE REINIGUNGSARTEN // ========================================== const stunden = parseFloat( document.getElementById('stunden').value ) || 0; const arbeitskosten = stunden * stundensatz; const flaechenkosten = flaeche * qmPreis; let netto = arbeitskosten + flaechenkosten + anfahrt + material; const zuschlagEuro = netto * zuschlagProzent / 100; netto += zuschlagEuro; const steuer = netto * mwst / 100; const brutto = netto + steuer; const box = document.getElementById('preisErgebnis'); box.style.display = 'block'; box.innerHTML = ` Berechnung

Reinigungsfläche: ${flaeche.toFixed(2)} m²
Arbeitsstunden: ${stunden.toFixed(2)} Std.
Arbeitskosten: ${arbeitskosten.toFixed(2)} €
Flächenkosten: ${flaechenkosten.toFixed(2)} €
Anfahrt: ${anfahrt.toFixed(2)} €
Material: ${material.toFixed(2)} €
Zuschlag: ${zuschlagEuro.toFixed(2)} €

Netto: ${netto.toFixed(2)} €
MwSt.: ${steuer.toFixed(2)} €
${brutto .toFixed(2) .replace('.', ',')} €
`; }
